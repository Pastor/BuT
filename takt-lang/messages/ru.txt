# Каталог сообщений: русский (базовый язык).
#
# Формат: `ключ = текст`, по строке на ключ; строки, начинающиеся с `#`, и
# пустые - пропускаются. Подстановка пишется как `{имя}` и приходит строкой:
# `{:?}` в каталог не попадает по построению (усиление фичи 0231).
# Перевод строки внутри текста записывается как `\n`.
#
# Русский каталог - база: набор ключей остальных языков сверяется с ним
# (гейт `scripts/check-messages.py`), и константы ключей `build.rs` строит
# тоже отсюда. Ключ - не код диагностики: у одного кода бывает несколько
# сообщений (см. `se-034.*`), и `with_code` остаётся литералом.

# ─── Подписи печатников диагностик (`diagnostics::batch`) ────────────────────
diag.compile-error = Ошибка компиляции [{code}]: {message}
diag.note-label = примечание
diag.warning = Предупреждение [{code}]: {message}

# ─── Отказ выбора языка (`diagnostics::lang`) ────────────────────────────────
lang.missing-value = ключ --lang требует кода языка
lang.unknown = неизвестный язык сообщений '{name}'; известны: {known}

# ─── Пробные коды задачи 0532-01 ─────────────────────────────────────────────
se-034.local-type-not-found = Локальный тип '{name}' не найден
se-034.parameter-type-unrecognised = тип параметра '{name}' не распознан: ожидались имя типа,                                          '[тип; N]' или 'q(m, n)'
