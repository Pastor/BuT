# ─────────────────────────────────────────────────────────────────────────────
# Сценарий: поездка с этажа 1 -> этаж 3 -> этаж 1
#
# Физическая модель датчиков:
#   floor_sensor_fN_bottom = 1, когда кабина входит в зону этажа N снизу.
#   floor_sensor_fN_top    = 1, когда кабина достигает верхней границы зоны.
#   Кнопки активны ровно один такт (импульс от пассажира).
#   Концевые датчики (elevator_motor_sensor_u/d) = защита от перехода, не задействованы.
#
# Ожидаемое поведение:
#   Тт 1-2 : Кабина на этаже 1, дверь открыта (Cabin=Idle, Motor=Idle)
#   Тт 3   : Нажата кнопка кабины "3" -> target_floor=3
#   Тт 4   : Cabin переходит в Moving, Motor начинает движение вверх
#   Тт 5-6 : Движение вверх, проезд зоны 2
#   Тт 7-8 : Движение вверх, проезд зоны 3
#   Тт 9   : Кабина в зоне этажа 3, target=current -> Cabin=AtFloor, Motor=Stop
#   Тт10-11: На этаже 3, дверь открыта
#   Тт 12  : Нажата кнопка "закрыть дверь" (DC) -> Cabin переходит в Idle
#   Тт 13  : Idle на этаже 3, нажата кнопка вызова на этаже 1 (floor_button_f1)
#   Тт14-15: Движение вниз, проезд зоны 2
#   Тт16-17: Движение вниз
#   Тт 18  : Кабина в зоне этажа 1, target=current -> AtFloor, Motor=Stop
#   Тт19-20: На этаже 1, дверь открыта
#   Тт 21  : Нажата кнопка "закрыть дверь" -> Cabin=Idle
#   Тт 22  : Простой, кабина на 1-м этаже, дверь открыта
# ─────────────────────────────────────────────────────────────────────────────

# == Такты 1-2: Кабина стоит на 1-м этаже ==
floor_sensor_f1_bottom=1
floor_sensor_f1_bottom=1

# == Такт 3: Пассажир нажимает кнопку "3" в кабине ==
floor_sensor_f1_bottom=1 cabin_button_f3=1

# == Такт 4: Дверь закрывается, кабина начинает движение вверх ==
# (датчики этажа 1 пока ещё активны, потом кабина уходит)
floor_sensor_f1_bottom=1

# == Такты 5-6: Движение вверх, между этажами 1 и 2 ==
# (нет активных датчиков этажей)

# (пустая строка = такт без активных датчиков)

# == Такт 7: Вход в зону этажа 2 ==
floor_sensor_f2_bottom=1

# == Такт 8: Проезд зоны 2 (Top-датчик) ==
floor_sensor_f2_top=1

# == Такт 9: Вход в зону этажа 3 -> совпадение с target (3) -> AtFloor ==
floor_sensor_f3_bottom=1

# == Такты 10-11: Кабина на этаже 3, дверь открыта ==
floor_sensor_f3_bottom=1 floor_sensor_f3_top=1
floor_sensor_f3_bottom=1 floor_sensor_f3_top=1

# == Такт 12: Пассажир нажимает кнопку "закрыть дверь" ==
floor_sensor_f3_bottom=1 cabin_button_dc=1

# == Такт 13: Idle на 3-м этаже; пассажир на 1-м этаже нажимает вызов ==
floor_sensor_f3_bottom=1 floor_button_f1=1

# == Такт 14: Кабина уходит вниз, покидает зону 3 ==
floor_sensor_f3_bottom=1

# == Такт 15: Между 3-м и 2-м ==

# == Такт 16: Вход в зону этажа 2 (снизу, при движении вниз = Top-датчик первый) ==
floor_sensor_f2_top=1

# == Такт 17: Проход зоны 2 ==
floor_sensor_f2_bottom=1

# == Такт 18: Вход в зону этажа 1 -> target=1=current -> AtFloor ==
floor_sensor_f1_bottom=1

# == Такты 19-20: На 1-м этаже, дверь открыта ==
floor_sensor_f1_bottom=1 floor_sensor_f1_top=1
floor_sensor_f1_bottom=1

# == Такт 21: Пассажир нажимает "закрыть дверь" ==
floor_sensor_f1_bottom=1 cabin_button_dc=1

# == Такт 22: Простой на 1-м этаже ==
floor_sensor_f1_bottom=1
